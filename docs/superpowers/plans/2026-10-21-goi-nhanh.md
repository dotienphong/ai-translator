```rust
/// Họ mẫu prompt của model dịch, theo trường `prompt` của file `mt` trong manifest (spec gói Nhanh §4.2).
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
pub enum PromptFamily {
    #[default]
    HyMt2,
    Lmt,
}

impl PromptFamily {
    pub fn from_code(code: &str) -> Option<Self> {
        match code {
            "hy-mt2" => Some(Self::HyMt2),
            "lmt" => Some(Self::Lmt),
            _ => None,
        }
    }

    pub fn code(self) -> &'static str {
        match self {
            Self::HyMt2 => "hy-mt2",
            Self::Lmt => "lmt",
        }
    }
}

/// Loại prompt của một câu (§6.5): thường, có thuật ngữ, hay có ngữ cảnh câu trước.
#[derive(Clone, Copy, Debug)]
pub enum PromptKind<'a> {
    Plain,
    Terminology(&'a [Term]),
    Context(&'a str),
}

/// Mẫu của model card `NiuTrans/LMT-60-0.6B` (revision `dd18984`): một lượt `user`, có ngôn ngữ nguồn.
fn lmt_prompt(text: &str, src: Lang, tgt: Lang) -> String {
    let (s, t) = (src.english_name(), tgt.english_name());
    format!("Translate the following text from {s} into {t}:\n{s}: {text}\n{t}:")
}

/// Điểm vào duy nhất cho nơi dịch: chọn mẫu theo họ và loại. LMT không có mẫu ngữ cảnh (spec gói Nhanh §5.3); mẫu thuật
/// ngữ của LMT theo kết quả spike (§6.1, Task 8 của kế hoạch).
pub fn build(family: PromptFamily, kind: PromptKind<'_>, text: &str, src: Lang, tgt: Lang) -> String {
    match (family, kind) {
        (PromptFamily::HyMt2, PromptKind::Plain) => translation_prompt(text, src, tgt),
        (PromptFamily::HyMt2, PromptKind::Terminology(terms)) => terminology_prompt(text, terms, src, tgt),
        (PromptFamily::HyMt2, PromptKind::Context(context)) => context_prompt(text, context, src, tgt),
        (PromptFamily::Lmt, PromptKind::Terminology(_) | PromptKind::Context(_) | PromptKind::Plain) => {
            lmt_prompt(text, src, tgt)
        }
    }
}
```

import json, sys, time, subprocess, urllib.request, unicodedata
srv, gguf, style, terms_path, outp = sys.argv[1:6]
PORT = 18432
EN = {"en": "English", "vi": "Vietnamese", "zh": "Chinese", "ja": "Japanese", "ko": "Korean"}
ZH = {"en": "英语", "vi": "越南语", "zh": "中文", "ja": "日语", "ko": "韩语"}

def hymt(r):
    s, t = r["src_lang"], r["tgt_lang"]
    if "zh" in (s, t):
        p = "参考下面的翻译：\n" + "".join(f"{x['source']} 翻译成 {x['target']}\n" for x in r["terms"])
        return p + f"将以下文本翻译为{ZH[t]}，注意只需要输出翻译后的结果，不要额外解释：\n\n{r['src']}"
    p = "Reference the following translations:\n" + "".join(f"{x['source']} translates to {x['target']}\n" for x in r["terms"])
    return p + (f"\nTranslate the following text into {EN[t]}. Note that you must ONLY output the translated result without "
                f"any additional explanation:\n\n{r['src']}")

def lmt_plain(r):
    s, t = EN[r["src_lang"]], EN[r["tgt_lang"]]
    return f"Translate the following text from {s} into {t}:\n{s}: {r['src']}\n{t}:"

def lmt(r):
    p = "Reference the following translations:\n" + "".join(f"{x['source']} translates to {x['target']}\n" for x in r["terms"])
    return p + "\n" + lmt_plain(r)

PROMPT = {"hymt": hymt, "lmt": lmt, "lmt-plain": lmt_plain}[style]
norm = lambda s: unicodedata.normalize("NFC", s).casefold()
p = subprocess.Popen([srv, "-m", gguf, "--port", str(PORT), "--host", "127.0.0.1", "-c", "2048", "-np", "1", "-ngl", "99"],
                     stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
try:
    for _ in range(240):
        try:
            if json.load(urllib.request.urlopen(f"http://127.0.0.1:{PORT}/health", timeout=1)).get("status") == "ok":
                break
        except Exception:
            pass
        time.sleep(0.5)
    rows = [json.loads(l) for l in open(terms_path)]
    hit = 0
    with open(outp, "w") as f:
        for r in rows:
            body = json.dumps({"messages": [{"role": "user", "content": PROMPT(r)}], "temperature": 0,
                               "repeat_penalty": 1.05, "max_tokens": 512}).encode()
            req = urllib.request.Request(f"http://127.0.0.1:{PORT}/v1/chat/completions", body, {"Content-Type": "application/json"})
            hyp = json.load(urllib.request.urlopen(req, timeout=120))["choices"][0]["message"]["content"].strip()
            ok = all(norm(x["target"]) in norm(hyp) for x in r["terms"])
            hit += ok
            f.write(json.dumps({"id": r["id"], "hyp": hyp, "all_terms": ok}, ensure_ascii=False) + "\n")
    print(f"{style}: đủ thuật ngữ {hit}/{len(rows)} = {100 * hit / len(rows):.1f}%")
finally:
