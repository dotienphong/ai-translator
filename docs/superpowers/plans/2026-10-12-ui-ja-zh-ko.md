export type { MessageKey };
export type UiLanguage = "en" | "vi" | "ja" | "zh" | "ko";
export type Params = Record<string, string | number>;

// Thứ tự trong ô chọn ngôn ngữ giao diện (bước 1 onboarding và Cài đặt › Chung); nhãn là `lang.<mã>`, tên mỗi ngôn ngữ
// viết bằng chính nó.
export const UI_LANGUAGES: readonly UiLanguage[] = ["vi", "en", "ja", "zh", "ko"];

const dictionaries: Record<UiLanguage, Record<MessageKey, string>> = { en, vi, ja, zh, ko };

// Đổi ngôn ngữ có tác dụng ngay (§4.5): giao diện gọi hàm này mỗi lần vẽ, với ngôn ngữ đang chọn.
export function translate(lang: UiLanguage, key: MessageKey, params?: Params): string {
  const template = dictionaries[lang][key];
  if (!params) return template;
  return template.replace(/\{(\w+)\}/g, (match, name: string) => (name in params ? String(params[name]) : match));
}

// Phần ngôn ngữ chính của locale → ngôn ngữ giao diện. Mọi locale tiếng Trung (kể cả phồn thể) ra `zh`, giao diện chữ
// giản thể (spec 2026-10-12 Q1); `yue` là tiếng Quảng Đông mà macOS báo là `yue-Hant-HK`.
const UI_BY_LOCALE = new Map<string, UiLanguage>([
  ["vi", "vi"],
  ["ja", "ja"],
  ["ko", "ko"],
  ["zh", "zh"],
  ["yue", "zh"],
]);

// Ngôn ngữ giao diện trước khi đọc được cài đặt, theo locale đầu tiên (§4.1, bước 1); không khớp thì English. Sau đó phía
// Rust quyết định theo locale của hệ điều hành (`ui_language_from_locale`, cùng bảng ví dụ).
export function detectUiLanguage(locales: readonly string[]): UiLanguage {
  const primary = (locales[0] ?? "").split(/[-_]/)[0]?.toLowerCase() ?? "";
  return UI_BY_LOCALE.get(primary) ?? "en";
}

// Giá trị thuộc tính `lang` của trang theo ngôn ngữ giao diện. WebView chọn font CJK theo thuộc tính này (chữ Hán Nhật và
// Hán Trung có hình khác nhau) và trình đọc màn hình chọn giọng đọc.
export function htmlLang(lang: UiLanguage): string {
  return lang === "zh" ? "zh-Hans" : lang;
}

// Khóa câu báo lỗi cho mã lỗi của phía Rust (`CommandError.code`); mã lạ thì dùng câu chung.
export function errorKey(code: string): MessageKey {
  const key = `error.${code}`;
  return key in en ? (key as MessageKey) : "error.unknown";

// Văn bản pháp lý đóng gói trong app (docs/legal/*.md; spec 2026-10-06 legal-in-app). Vite đóng mỗi file thành một
// chunk riêng, chỉ tải khi người dùng mở văn bản. Theo mẫu của notices.ts.
import type { UiLanguage } from "../i18n";

export type LegalKind = "eula" | "privacy";

/** Kết quả của `import.meta.glob(…, { query: "?raw", import: "default" })`: đường dẫn → hàm tải văn bản. */
export type LegalLoaders = Record<string, () => Promise<string>>;

/** Văn bản đã tải, kèm ngôn ngữ của chính văn bản (có thể khác ngôn ngữ giao diện). */
export interface LegalText {
  text: string;
  lang: UiLanguage;
}

const docs = import.meta.glob<string>("/docs/legal/{eula,privacy}.{vi,en}.md", { query: "?raw", import: "default" });

/**
 * Hàm tải văn bản đúng loại, kèm ngôn ngữ của file tìm được: bản đúng ngôn ngữ giao diện; không có thì bản English (ja, zh,
 * ko chưa có bản pháp lý riêng, spec 2026-10-12 Q2); không có nữa thì bản tiếng Việt (bản gốc); không có thì null.
 */
export function pickLegal(
  loaders: LegalLoaders,
  kind: LegalKind,
  lang: UiLanguage,
): { load: () => Promise<string>; lang: UiLanguage } | null {
  for (const candidate of [lang, "en", "vi"] as const) {
    const found = Object.entries(loaders).find(([path]) => path.endsWith(`/${kind}.${candidate}.md`));
    if (found) return { load: found[1], lang: candidate };
  }
  return null;
}

/** Văn bản và ngôn ngữ của nó, hoặc null khi không có (hay file rỗng, hay tải lỗi). */
export async function loadLegal(kind: LegalKind, lang: UiLanguage, loaders: LegalLoaders = docs): Promise<LegalText | null> {
  const picked = pickLegal(loaders, kind, lang);
  if (!picked) return null;
  try {
    const text = await picked.load();
    return text.trim() === "" ? null : { text, lang: picked.lang };
  } catch {
    return null;
  }
}

  await page.goto(url);
  await sleep(800);
  if (prepare) {
    await page.eval(prepare);
    await sleep(wait);
  }
  await check(label, page, `document.documentElement.lang === ${JSON.stringify(HTML_LANG[lang])}`, `<html lang> phải là ${HTML_LANG[lang]}`);
  if (audit) {
    const issues = await page.eval(AUDIT).catch((e) => [`lỗi: ${e.message}`]);
    for (const issue of issues) failures.push(`${label}: ${issue}`);
